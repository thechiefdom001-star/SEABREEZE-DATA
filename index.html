const SHEET_NAME = 'Customers';
function doGet(e){
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const values = sheet.getDataRange().getValues();
  const headers = values.shift();
  const rows = values.map(r=>{let o={};headers.forEach((h,i)=>o[h]=r[i]);return o;});
  return ContentService.createTextOutput(JSON.stringify({headers,rows})).setMimeType(ContentService.MimeType.JSON);
}
function doPost(e){
  const data = JSON.parse(e.postData.contents);
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  sheet.clearContents();
  const headers = data.headers;
  sheet.appendRow(headers);
  data.rows.forEach(r=>sheet.appendRow(headers.map(h=>r[h]||'')));
  return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
}
